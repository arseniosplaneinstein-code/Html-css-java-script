  <script>
    // Mostrar TechBot após 3 segundos
    setTimeout(() => {
      document.getElementById('techbot').style.display = 'block';
    }, 3000);

    // Logo pequena aparece no canto esquerdo quando rolar
    window.addEventListener("scroll", function() {
      const header = document.getElementById("header");
      const logoSmall = document.getElementById("logoSmall");

      if (window.scrollY > 150) {
        header.style.opacity = "0";
        header.style.height = "0";
        logoSmall.style.opacity = "1";
        logoSmall.style.transform = "translateX(0)";
      } else {
        header.style.opacity = "1";
        header.style.height = "auto";
        logoSmall.style.opacity = "0";
        logoSmall.style.transform = "translateX(-50px)";
      }
    });

    // Funções do TechBot
    function techbotShowSite() {
      const botText = document.getElementById('techbot-text');
      botText.innerHTML = "🚀 O site tem várias áreas incríveis! <br><br> - Início: apresenta o tema.<br> - Robótica: explica o que é e como funciona.<br> - Serviços e Galeria: mostram projetos e imagens.<br><br> Quer aprender sobre os robôs agora?";
      document.getElementById('techbot-buttons').innerHTML = `
        <button onclick="techbotExplainRobot()">Sim 🧠</button>
        <button onclick="techbotBye()">Mais tarde 👋</button>
      `;
    }

    function techbotExplainRobot() {
      const botText = document.getElementById('techbot-text');
      botText.innerHTML = "🤖 Um robô é uma máquina programável feita de sensores, motores e controladores.<br><br> Ele é construído com componentes de hardware e software que o ajudam a perceber, pensar e agir. A robótica usa áreas como eletrónica, mecânica e programação para dar vida às máquinas!";
      document.getElementById('techbot-buttons').innerHTML = `
        <button onclick="techbotBye()">Legal! 👋</button>
      `;
    }

    function techbotBye() {
      const bot = document.getElementById('techbot');
      bot.style.display = 'none';
    }
